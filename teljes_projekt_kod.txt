// HitelesÃ­tÃ©si kÃ¶ztesrÃ©teg (Authentication Middleware)
// Feladata a bejÃ¶vÅ‘ HTTP kÃ©rÃ©sek fejlÃ©cÃ©ben talÃ¡lhatÃ³ JWT token Ã©rvÃ©nyesÃ­tÃ©se

const jwt = require('jsonwebtoken');

module.exports = function (req, res, next) {
    // 1. Token kinyerÃ©se a kÃ©rÃ©s fejlÃ©cÃ©bÅ‘l (Authorization header)
    const token = req.header('Authorization');

    // 2. Ha nincs token, a hozzÃ¡fÃ©rÃ©s megtagadÃ¡sa
    if (!token) {
        return res.status(401).json({ message: 'Nincs token megadva, a hitelesÃ­tÃ©s megtagadva.' });
    }

    try {
        // 3. A "Bearer " elÅ‘tag levÃ¡gÃ¡sa a token stringrÅ‘l, amennyiben jelen van
        const actualToken = token.startsWith('Bearer ') ? token.slice(7, token.length).trimLeft() : token;

        // 4. Token dekÃ³dolÃ¡sa Ã©s ellenÅ‘rzÃ©se a titkos kulcs segÃ­tsÃ©gÃ©vel
        const decoded = jwt.verify(actualToken, process.env.JWT_SECRET);
        
        // 5. A dekÃ³dolt felhasznÃ¡lÃ³i adatok (ID, role) csatolÃ¡sa a kÃ©rÃ©s (req) objektumhoz
        req.user = decoded;
        
        // 6. VezÃ©rlÃ©s Ã¡tadÃ¡sa a kÃ¶vetkezÅ‘ middleware-nek vagy a vÃ©gsÅ‘ ÃºtvonalkezelÅ‘nek
        next();
    } catch (error) {
        res.status(401).json({ message: 'Ã‰rvÃ©nytelen vagy lejÃ¡rt token.' });
    }
};

// [GET] Szabad idÅ‘pontok lekÃ©rÃ©se egy adott napra, munkatÃ¡rsra Ã©s szolgÃ¡ltatÃ¡sra
router.get('/available-slots', async (req, res) => {
    try {
        const { staffId, date, serviceId } = req.query;
        if (!staffId || !date || !serviceId) return res.status(400).json({ message: 'HiÃ¡nyzÃ³ paramÃ©terek.' });

        const staff = await Staff.findById(staffId);
        const service = await Service.findById(serviceId);
        if (!staff || !service) return res.status(404).json({ message: 'MunkatÃ¡rs vagy szolgÃ¡ltatÃ¡s nem talÃ¡lhatÃ³.' });

        // Keresett nap sorszÃ¡mÃ¡nak illesztÃ©se (1: HÃ©tfÅ‘ ... 7: VasÃ¡rnap)
        const queryDate = new Date(date);
        const jsDay = queryDate.getDay();
        const dayOfWeek = jsDay === 0 ? 7 : jsDay; 

        // MunkatÃ¡rs aznapi munkaideje
        const workDay = staff.availability.workingHours.find(wh => wh.dayOfWeek === dayOfWeek);
        if (!workDay) return res.status(200).json([]); // Ha nincs talÃ¡lat, nem dolgozik ezen a napon

        // Aznapi lÃ©tezÅ‘, aktÃ­v foglalÃ¡sok lekÃ©rÃ©se (elutasÃ­tottakat nem nÃ©zzÃ¼k)
        const existingBookings = await Booking.find({
            staffId,
            date: date,
            status: { $ne: 'cancelled' }
        });

        let availableSlots = [];
        let currentSlot = new Date(`${date}T${workDay.startTime}:00`);
        const endOfDay = new Date(`${date}T${workDay.endTime}:00`);
        const slotInterval = staff.availability.slotInterval || 30; // 30 perces ugrÃ¡sokkal haladunk

        // IdÅ‘pontok generÃ¡lÃ¡sa Ã©s Ã¼tkÃ¶zÃ©svizsgÃ¡lat
        while (currentSlot < endOfDay) {
            const slotStart = new Date(currentSlot);
            const slotEnd = new Date(currentSlot.getTime() + (service.duration * 60000));

            // Ha a szolgÃ¡ltatÃ¡s vÃ©ge tÃºlnyÃºlna a munkaidÅ‘n, megszakÃ­tjuk a keresÃ©st
            if (slotEnd > endOfDay) break;

            // EllenÅ‘rizzÃ¼k, hogy a generÃ¡lt idÅ‘pont Ã¼tkÃ¶zik-e bÃ¡rmelyik meglÃ©vÅ‘ foglalÃ¡ssal
            const isOverlapping = existingBookings.some(booking => {
                const bStart = new Date(`${date}T${booking.startTime}:00`);
                const bEnd = new Date(`${date}T${booking.endTime}:00`);
                return (slotStart < bEnd && slotEnd > bStart);
            });

            if (!isOverlapping) {
                const timeString = slotStart.toTimeString().substring(0, 5); // "HH:MM" formÃ¡tum
                availableSlots.push(timeString);
            }

            // UgrÃ¡s a kÃ¶vetkezÅ‘ slotra (pl. 08:00 -> 08:30)
            currentSlot = new Date(currentSlot.getTime() + (slotInterval * 60000));
        }

        res.status(200).json(availableSlots);
    } catch (error) {
        console.error(error);
        res.status(500).json({ message: 'Hiba a szabad idÅ‘pontok szÃ¡mÃ­tÃ¡sakor.' });
    }
});

import React from 'react';

const isTokenValid = (token) => {
    if (!token) return false;
    try {
        const payloadBase64 = token.split('.')[1];
        const decodedPayload = JSON.parse(atob(payloadBase64));
        if (decodedPayload.exp && decodedPayload.exp * 1000 < Date.now()) return false;
        return true;
    } catch (error) {
        return false;
    }
};

// Ãšj paramÃ©ter: allowedRoles (engedÃ©lyezett szerepkÃ¶rÃ¶k listÃ¡ja)
const ProtectedRoute = ({ children, allowedRoles }) => {
    const token = localStorage.getItem('token');
    const userStr = localStorage.getItem('user');

    if (!isTokenValid(token) || !userStr) {
        localStorage.removeItem('token');
        localStorage.removeItem('user');
        return <Navigate to="/login" replace />;
    }

    const user = JSON.parse(userStr);
    const userRole = user.role || 'user'; // AlapÃ©rtelmezett vendÃ©g szerepkÃ¶r

    // Ha a route vÃ©dett (pl. csak cÃ©ges), Ã©s a felhasznÃ¡lÃ³ szerepkÃ¶re nincs engedÃ©lyezve
    if (allowedRoles && !allowedRoles.includes(userRole)) {
        if (userRole === 'user' || userRole === 'guest') {
            return <Navigate to="/vendeg-profil" replace />;
        }
        return <Navigate to="/" replace />;
    }

    return children;
};

function App() {

    // TÃ©ma vÃ¡ltÃ¡sa Ã©s elmentÃ©se a bÃ¶ngÃ©szÅ‘ memÃ³riÃ¡jÃ¡ba
    const toggleColorMode = () => {
        setMode((prevMode) => {

            return newMode;

    };

    const theme = useMemo(() => createTheme({
        palette: {

    return (

                    <AppContent toggleColorMode={toggleColorMode} mode={mode} />
                </Router>

    );
}
